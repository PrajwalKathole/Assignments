app.py
app.py
from flask import Flask

app = Flask(__name__)

@app.route("/")
def hello():
    return "Hello World from Flask in Docker!"

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)


DockerFile for Flask
FROM python:3.11-slim

WORKDIR /app

COPY app.py .

RUN    pip     install     flask

EXPOSE     5000

CMD     ["python",  "app.py"]
