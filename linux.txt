#!/bin/bash

# Read first number with default value … -e means arrow keys, backspace etc allowed, since we are providing default input values that user may want to edit
read -e -i "10" -p "Enter first number: " a
echo

# Read second number with default value
read -e -i "5" -p "Enter second number: " b
echo

# Read operation with default value
read -e -i "+" -p "Enter operation (+, -, *, /): " op
echo

# Case statement
case $op in
    +) echo "Result = $((a + b))" ;;
    -) echo "Result = $((a - b))" ;;
    \*) echo "Result = $((a * b))" ;;
    /) echo "Result = $((a / b))" ;;
    *) echo "Invalid operation" ;;
esac
